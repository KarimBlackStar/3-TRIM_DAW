📚 Parte 1 — BIBLIOTECA
1

Vista v_socio_presta

Socios + nº préstamos · ordenado DESC · 0,5 pts

CREATE OR REPLACE VIEW v_socio_presta AS
SELECT
    s.nsocio,
    s.nombre,
    s.apellidos,
    COUNT(p.nprestamo) AS num_prestamos
FROM SOCIOS s
LEFT JOIN PRESTAMOS p
    ON s.nsocio = p.nsocio
GROUP BY s.nsocio, s.nombre, s.apellidos
ORDER BY num_prestamos DESC;
🔗
LEFT JOIN: incluye socios SIN préstamos (contador = 0). Con INNER JOIN se perderían.
🚫
NO es actualizable: usa COUNT() + GROUP BY → solo lectura. No se puede INSERT/UPDATE/DELETE.
2

Vista v_nuevo_socio

Alta socios con validación provincia/CP · 0,5 pts

CREATE OR REPLACE VIEW v_nuevo_socio AS
SELECT nsocio, nombre, apellidos,
       direccion, c_postal, poblacion,
       provincia, telefono
FROM SOCIOS
WHERE provincia BETWEEN '01' AND '52'
  AND LEFT(c_postal, 2) = provincia
WITH CHECK OPTION;
✅
WITH CHECK OPTION: obliga a que INSERT/UPDATE cumplan el WHERE. Si no, MySQL lanza error.
📮
LEFT(c_postal, 2): extrae los 2 primeros dígitos del CP y los compara con la provincia.
✍️
SÍ es actualizable: sin GROUP BY ni funciones de agregado → permite INSERT/UPDATE.
🛒 Parte 2 — VENTAS_TIENDAS
3

Función fn_ventas

Total artículos vendidos en una tienda · 0,5 pts

DELIMITER $$

CREATE FUNCTION fn_ventas(p_tda_num INT)
RETURNS INT
READS SQL DATA
BEGIN
    DECLARE total INT;

    SELECT IFNULL(SUM(ven_cant), 0)
    INTO total
    FROM VENTAS
    WHERE ven_tda = p_tda_num;

    RETURN total;
END$$

DELIMITER ;
0️⃣
IFNULL(SUM(...), 0): si la tienda no existe, SUM devuelve NULL → lo convierte en 0.
🔄
NO es determinística: mismos parámetros pueden dar resultados distintos (la tabla cambia). Por eso: READS SQL DATA, no DETERMINISTIC.
4

Procedimiento act_ventas

Actualiza clt_ventas de todos los clientes · 1 pt

DELIMITER $$

CREATE PROCEDURE act_ventas()
BEGIN
    UPDATE CLIENTES c
    SET clt_ventas = (
        SELECT COUNT(*)
        FROM VENTAS v
        WHERE v.ven_clt = c.clt_num
    );
END$$

DELIMITER ;
🔁
Subconsulta correlacionada: por cada fila de CLIENTES, cuenta sus ventas en VENTAS. Muy eficiente.
⚡
El UPDATE modifica TODOS los clientes de golpe, sin bucles.
5

Procedimiento ventas_fecha

Ventas entre dos fechas · 0,5 pts

DELIMITER $$

CREATE PROCEDURE ventas_fecha(
    p_fecha_ini DATE,
    p_fecha_fin DATE
)
BEGIN
    SELECT
        v.ven_tda     AS tienda,
        v.ven_fecha   AS fecha,
        CONCAT(c.clt_nom,' ',c.clt_apel)
                      AS cliente,
        v.ven_importe AS importe
    FROM VENTAS v
    JOIN CLIENTES c
        ON v.ven_clt = c.clt_num
    WHERE v.ven_fecha
        BETWEEN p_fecha_ini
        AND p_fecha_fin
    ORDER BY v.ven_fecha;
END$$

DELIMITER ;
📅
BETWEEN: es inclusivo en ambos extremos (incluye fecha inicio y fecha fin).
👤
CONCAT: une nombre y apellidos del cliente en una sola columna de resultado.
6

Disparadores INSERT/DELETE/UPDATE

Stock + clt_ventas al cambiar VENTAS · 1,5 pts

DELIMITER $$

CREATE TRIGGER trg_ventas_after_insert
AFTER INSERT ON VENTAS
FOR EACH ROW
BEGIN
  -- Reducir stock
  UPDATE ARTICULOS
  SET art_stock = art_stock - NEW.ven_cant
  WHERE art_num = NEW.ven_art;

  -- Incrementar contador cliente
  UPDATE CLIENTES
  SET clt_ventas = clt_ventas + 1
  WHERE clt_num = NEW.ven_clt;
END$$

DELIMITER ;
DELIMITER $$

CREATE TRIGGER trg_ventas_after_delete
AFTER DELETE ON VENTAS
FOR EACH ROW
BEGIN
  -- Devolver stock
  UPDATE ARTICULOS
  SET art_stock = art_stock + OLD.ven_cant
  WHERE art_num = OLD.ven_art;

  -- Decrementar contador cliente
  UPDATE CLIENTES
  SET clt_ventas = clt_ventas - 1
  WHERE clt_num = OLD.ven_clt;
END$$

DELIMITER ;
DELIMITER $$

CREATE TRIGGER trg_ventas_after_update
AFTER UPDATE ON VENTAS
FOR EACH ROW
BEGIN
  -- Ajustar stock (mismo artículo)
  UPDATE ARTICULOS
  SET art_stock =
    art_stock + OLD.ven_cant - NEW.ven_cant
  WHERE art_num = OLD.ven_art;

  -- Si cambió el artículo
  IF OLD.ven_art != NEW.ven_art THEN
    UPDATE ARTICULOS
    SET art_stock = art_stock + OLD.ven_cant
    WHERE art_num = OLD.ven_art;

    UPDATE ARTICULOS
    SET art_stock = art_stock - NEW.ven_cant
    WHERE art_num = NEW.ven_art;
  END IF;

  -- Si cambió el cliente
  IF OLD.ven_clt != NEW.ven_clt THEN
    UPDATE CLIENTES
    SET clt_ventas = clt_ventas - 1
    WHERE clt_num = OLD.ven_clt;

    UPDATE CLIENTES
    SET clt_ventas = clt_ventas + 1
    WHERE clt_num = NEW.ven_clt;
  END IF;
END$$

DELIMITER ;
🆕
NEW: valores tras el cambio. Solo en INSERT y UPDATE.
🗂️
OLD: valores anteriores. Solo en DELETE y UPDATE.
⚠️
En UPDATE: si cambia el artículo, hay que ajustar el stock de AMBOS artículos.
7

Disparador stock negativo

Impide stock < 0 al insertar venta · 1 pt

DELIMITER $$

CREATE TRIGGER trg_check_stock_before_insert
BEFORE INSERT ON VENTAS
FOR EACH ROW
BEGIN
  DECLARE stock_actual INT;

  SELECT art_stock INTO stock_actual
  FROM ARTICULOS
  WHERE art_num = NEW.ven_art;

  IF stock_actual - NEW.ven_cant < 0 THEN
    SIGNAL SQLSTATE '45000'
    SET MESSAGE_TEXT =
      'Error: El stock no puede ser negativo';
  END IF;
END$$

DELIMITER ;
⏮️
BEFORE INSERT: se ejecuta ANTES de guardar → puede cancelar. Con AFTER ya sería tarde.
🚨
SIGNAL SQLSTATE '45000': estándar MySQL para errores personalizados desde triggers/procedimientos.
8

Evento limpieza ventas

Borra ventas >1 año · cada año · 4 años · 0,5 pts

CREATE EVENT evt_limpiar_ventas
ON SCHEDULE EVERY 1 YEAR
    STARTS '2026-01-01 00:00:00'
    ENDS   '2029-01-01 00:00:00'
ON COMPLETION PRESERVE
DO
    DELETE FROM VENTAS
    WHERE ven_fecha <
        DATE_SUB(CURDATE(),
            INTERVAL 1 YEAR);
📆
EVERY 1 YEAR: 4 ejecuciones → 2026, 2027, 2028, 2029.
💾
ON COMPLETION PRESERVE: al acabar, el evento queda desactivado pero NO se borra. Sin esto, MySQL lo elimina automáticamente.
📉
DATE_SUB(CURDATE(), INTERVAL 1 YEAR): calcula la fecha de hace exactamente un año.
